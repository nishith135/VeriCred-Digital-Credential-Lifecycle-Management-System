  function inspectTrace(id, title, latency, traceText) {
    document.getElementById('modalTitle').textContent = 'Engine Trace: ' + title;
    document.getElementById('modalSimId').textContent = id;
    document.getElementById('modalLatency').textContent = latency;
    document.getElementById('modalTraceContent').innerHTML = traceText;
    document.getElementById('traceModal').classList.remove('hidden');
  }

  function closeTrace() {
    document.getElementById('traceModal').classList.add('hidden');
  }

  function clearConsole() {
    const el = document.getElementById('telemetryLog');
    el.innerHTML = '<div class="text-outline">[CLEARED] Awaiting live telemetry triggers...</div>';
  }

  function appendLog(line, isHighlight) {
    const el = document.getElementById('telemetryLog');
    const d = new Date();
    const timeStr = '[' + d.toTimeString().split(' ')[0] + '.' + String(d.getMilliseconds()).padStart(3, '0') + ']';
    const div = document.createElement('div');
    if (isHighlight) {
      div.className = 'text-primary font-semibold pt-1';
      div.innerHTML = '<span class="text-outline/70">' + timeStr + '</span> ' + line;
    } else {
      div.className = 'text-primary-container';
      div.innerHTML = '<span class="text-outline/70">' + timeStr + '</span> ' + line;
    }
    el.appendChild(div);
    el.scrollTop = el.scrollHeight;
  }

  function rerunSingleTest(id, baseLatency) {
    const nowLatency = (baseLatency + (Math.random() * 2 - 1)).toFixed(1);
    appendLog('<span class="text-secondary font-medium">[' + id + ']</span> Manual cycle requested: Invariant verified in ' + nowLatency + 'ms.');
  }

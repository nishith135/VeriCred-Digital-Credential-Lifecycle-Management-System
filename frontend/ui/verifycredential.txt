  function setChecklistState(f1, f2, f3, f4) {
    updateCard('factor-1', f1, 'Active in Registry', 'Record Non-Existent');
    updateCard('factor-2', f2, 'SHA-256 Match', 'Digest Mismatch');
    updateCard('factor-3', f3, 'Ed25519 Valid', 'Signature Invalid');
    updateCard('factor-4', f4, 'Unrevoked', 'Tombstone Active');
  }

  function updateCard(prefix, ok, okText, failText) {
    const verdict = document.getElementById(`${prefix}-verdict`);
    const icon = document.getElementById(`${prefix}-icon`);
    const bar = document.getElementById(`${prefix}-bar`);

    if (ok) {
      verdict.innerText = okText;
      verdict.className = 'font-headline-sm text-headline-sm text-primary font-semibold';
      icon.innerText = 'check_circle';
      icon.className = 'material-symbols-outlined text-[18px] text-primary';
      bar.className = 'h-full bg-primary-container w-full';
    } else {
      verdict.innerText = failText;
      verdict.className = 'font-headline-sm text-headline-sm text-error font-semibold';
      icon.innerText = 'cancel';
      icon.className = 'material-symbols-outlined text-[18px] text-error';
      bar.className = 'h-full bg-error w-full';
    }
  }

<script>
  function openRevokeModal(credId, learner, course) {
    if (credId) document.getElementById('modalCredId').textContent = credId;
    if (learner) document.getElementById('modalLearner').textContent = learner;
    if (course) document.getElementById('modalCourse').textContent = course;
    document.getElementById('revocationModal').classList.remove('hidden');
  }

  function closeRevokeModal() {
    document.getElementById('revocationModal').classList.add('hidden');
  }

  function showToast(message, isError = false) {
    const toast = document.getElementById('toastNotification');
    const toastMsg = document.getElementById('toastMessage');
    const toastIcon = document.getElementById('toastIcon');

    toastMsg.textContent = message;
    if (isError) {
      toastIcon.textContent = 'error';
      toastIcon.className = 'material-symbols-outlined text-[20px] text-error';
    } else {
      toastIcon.textContent = 'check_circle';
      toastIcon.className = 'material-symbols-outlined text-[20px] text-primary';
    }

    toast.classList.remove('translate-y-20', 'opacity-0');
    setTimeout(() => {
      toast.classList.add('translate-y-20', 'opacity-0');
    }, 4000);
  }

  function executeRevocation() {
    const check = document.getElementById('governanceCheck');
    if (!check.checked) {
      showToast('Mandatory governance verification checkbox must be acknowledged', true);
      return;
    }
    const credId = document.getElementById('modalCredId').textContent;
    const btn = document.getElementById('confirmRevokeBtn');
    btn.disabled = true;
    btn.innerHTML = '<span class="material-symbols-outlined animate-spin text-[18px]">progress_activity</span> Signing Tombstone...';

    setTimeout(() => {
      btn.disabled = false;
      btn.innerHTML = '<span class="material-symbols-outlined text-[18px]">gavel</span> Confirm Irreversible Revocation';
      closeRevokeModal();
      showToast(`Tombstone for ${credId} permanently committed to Block #1,492,085`);
    }, 1200);
  }

  function triggerSignAndCommit() {
    const btn = document.getElementById('issueSignBtn');
    btn.disabled = true;
    btn.innerHTML = '<span class="material-symbols-outlined animate-spin text-[18px]">progress_activity</span> Signing with Ed25519 Enclave...';

    setTimeout(() => {
      btn.disabled = false;
      btn.innerHTML = '<span class="material-symbols-outlined text-[18px]">check_circle</span> Credential Issued: CRED-2026-090412';
      btn.classList.remove('bg-primary');
      btn.classList.add('bg-surface-container-high', 'text-primary');
      showToast('Credential successfully signed and recorded in PostgreSQL 17 ledger pool');
    }, 1400);
  }

  function evaluateDirect(learnerName, stuId) {
    document.getElementById('evaluationDrawer').classList.remove('hidden');
    showToast(`Evaluating candidate rules for ${learnerName} (${stuId})`);
  }

  function showDetails(credId) {
    showToast(`Loading cryptographic attestation proof for ${credId}...`);
  }

  function openIssuanceModal() {
    document.getElementById('evaluationDrawer').classList.remove('hidden');
    document.getElementById('evaluationDrawer').scrollIntoView({ behavior: 'smooth' });
    showToast('Eligibility evaluation workflow deck opened for candidate review');
  }

  // Quick filter search simulation
  document.getElementById('credentialSearch').addEventListener('input', function(e) {
    const term = e.target.value.toLowerCase();
    const rows = document.querySelectorAll('tbody tr');
    rows.forEach(row => {
      const text = row.innerText.toLowerCase();
      row.style.display = text.includes(term) ? '' : 'none';
    });
  });

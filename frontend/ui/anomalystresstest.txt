<!-- Client-Side State Toggle Script -->
<script>
    function toggleDetailsRow(rowId) {
      const row = document.getElementById(rowId);
      const icon = document.getElementById('detail-icon-1');
      const label = document.getElementById('detail-label-1');
      
      if (!row) return;

      if (row.classList.contains('hidden')) {
        row.classList.remove('hidden');
        if (icon) icon.textContent = 'expand_less';
        if (label) label.textContent = 'Collapse';
      } else {
        row.classList.add('hidden');
        if (icon) icon.textContent = 'expand_more';
        if (label) label.textContent = 'Inspect';
      }
    }

    function switchLedgerState(state) {
      const table = document.getElementById('ledgerContentContainer');
      const loading = document.getElementById('ledgerLoadingState');
      const empty = document.getElementById('ledgerEmptyState');

      const btnTable = document.getElementById('btnStateTable');
      const btnLoading = document.getElementById('btnStateLoading');
      const btnEmpty = document.getElementById('btnStateEmpty');

      // Reset buttons
      [btnTable, btnLoading, btnEmpty].forEach(btn => {
        btn.className = "px-space-xs py-0.5 rounded text-outline hover:text-on-surface text-code-dense font-code-dense";
      });

      // Hide all
      table.classList.add('hidden');
      loading.classList.add('hidden');
      loading.classList.remove('flex');
      empty.classList.add('hidden');
      empty.classList.remove('flex');

      if (state === 'table') {
        table.classList.remove('hidden');
        btnTable.className = "px-space-xs py-0.5 rounded bg-surface-container-high text-on-surface text-code-dense font-code-dense";
      } else if (state === 'loading') {
        loading.classList.remove('hidden');
        loading.classList.add('flex');
        btnLoading.className = "px-space-xs py-0.5 rounded bg-surface-container-high text-on-surface text-code-dense font-code-dense";
      } else if (state === 'empty') {
        empty.classList.remove('hidden');
        empty.classList.add('flex');
        btnEmpty.className = "px-space-xs py-0.5 rounded bg-surface-container-high text-on-surface text-code-dense font-code-dense";
      }
    }

    function applyLedgerFilters() {
      // Micro visual confirmation for the analyst
      const container = document.getElementById('ledgerContentContainer');
      container.classList.add('opacity-50');
      setTimeout(() => {
        container.classList.remove('opacity-50');
      }, 180);
    }
  </script>
